FTtestComplete_test_=: 3 : 0
   0 FTtestComplete y.        NB. if 1=x., show details even if all tests passed.
:
   coinsert 'base'
   if. 0=#y. do. y.=. FTarg1 end.
   y.=. toupper&.>y.
   reconfls=. _1~:({.@:,)&>(1{"1 FT) rxmatch&.>/y.
   sumrecon=. +/reconfls
   try. FTtest sumrecon;<reconfls
       if. x. do. FTshowTests sumrecon;reconfls;<y. end.
   catch. FTshowTests sumrecon;reconfls;<y.
   end.
)

FTtest_test_=: 3 : 0
   'sumrecon reconfls'=. y.
NB. Unrecognized test strings: FTarg1_test_#~-.+./reconfls
   assert. sumrecon *./ . e. 1 0   NB. No more than 1 regex/match
   assert. 1*./ . <:+/|:reconfls   NB. All regexes tested once or more
   assert. 0 1 *./ . e.~ sumrecon  NB. No duplicate classification
   1
)

FTshowTests_test_=: 3 : 0
   'sumrecon reconfls args'=. y.

   ntre=. 0=+/|:reconfls                          NB. Untested regexes
   if. 0<pl=. +/ntre do. pl=. (1~:pl)#'es'
       smoutput 'Untested regex',pl,':'
       smoutput >(<'   '),&.>ntre#1{"1 FT
   end.

   retm1=. 1<+/|:reconfls                         NB. Regexes tested more than once
   if. 0<pl=. +/retm1 do. pl=. (1~:pl)#'es'
       smoutput 'Regex',pl,' tested more than once:'
       smoutput >(<'   '),&.>retm1#1{"1 FT
   end.

   dupcl=. -.0 1 e.~sumrecon                      NB. Duplicate classification
   if. 0<nd=. +/dupcl do.
       smoutput 'Duplicate classification',('s'#~1~:nd),':'
       for_dix. I. dupcl do.
           smoutput FT#~dix{"1 reconfls
           smoutput ''
       end.
   end.

   if. 0<pl=. +/0=sumrecon do. pl=. (1~:pl)#'s'   NB. Unclassified targets
       smoutput 'Unclassified target string',pl,':'
       smoutput >(<'   '),&.>(0=sumrecon)#args
   end.
)
